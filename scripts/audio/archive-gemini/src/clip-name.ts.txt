/** 53-bit string hash (cyrb53): enough to name a cached clip without collisions among the few thousand a phone keeps. */
function hash(text: string): string {
  let h1 = 0xdeadbeef;
  let h2 = 0x41c6ce57;
  for (let i = 0; i < text.length; i += 1) {
    const code = text.charCodeAt(i);
    h1 = Math.imul(h1 ^ code, 2654435761);
    h2 = Math.imul(h2 ^ code, 1597334677);
  }
  h1 = Math.imul(h1 ^ (h1 >>> 16), 2246822507) ^ Math.imul(h2 ^ (h2 >>> 13), 3266489909);
  h2 = Math.imul(h2 ^ (h2 >>> 16), 2246822507) ^ Math.imul(h1 ^ (h1 >>> 13), 3266489909);
  return (4294967296 * (2097151 & h2) + (h1 >>> 0)).toString(36);
}

/** The file name a spoken clip is kept under: the same words in the same voice are made once and played from the phone afterwards (offline too). Pure. */
export const clipFileName = (voice: string, text: string) => `${voice.toLowerCase()}-${hash(`${voice}\n${text}`)}.wav`;
